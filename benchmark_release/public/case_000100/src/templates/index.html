Quick Start

# 1. Authenticate
curl -s -c cookies.txt -X POST http://localhost:9000/login \
  -H 'Content-Type: application/json' \
  -d '{"username":"alice","password":"aliceS3cure!"}'

# 2. Start an import job
curl -s -b cookies.txt -X POST http://localhost:9000/api/images/import \
  -H 'Content-Type: application/json' \
  -d '{"image_name":"my-ubuntu","source_image":"projects/ubuntu-os-cloud/global/images/ubuntu-2204"}'
